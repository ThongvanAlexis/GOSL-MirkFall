import 'package:vector_tile_renderer/vector_tile_renderer.dart' as vtr;

/// Charge `assets/maps/style.json` et le compile en [vtr.Theme] via `ThemeReader`, une seule fois
/// par [MapTheme] (construire un Theme par build est l'anti-pattern n°1 du POC : churn de rebuild).
/// Remplace `StyleRewriter` : plus de réécriture d'URI, le chemin PMTiles est donné au provider.
class MapThemeLoader {
  MapThemeLoader({AssetBundle? bundle}) : _bundle = bundle ?? rootBundle;
  final AssetBundle _bundle;
  final Map<MapTheme, Future<vtr.Theme>> _themeFutureByTheme = <MapTheme, Future<vtr.Theme>>{};

  /// Chemin d'asset par thème. `rpgParchment` pointe sur le style standard jusqu'à la Phase 13
  /// (second JSON) — `setTheme` n'a aucun appelant aujourd'hui (RESEARCH Open Question 7).
  static String assetPathFor(MapTheme theme) => switch (theme) {
    MapThemeStandard() => kStyleJsonAssetPath,
    MapThemeRpgParchment() => kStyleJsonAssetPath,
  };

  Future<vtr.Theme> load(MapTheme theme) => _themeFutureByTheme.putIfAbsent(theme, () => _loadUncached(theme));

  Future<vtr.Theme> _loadUncached(MapTheme theme) async {
    final String raw = await _bundle.loadString(assetPathFor(theme));
    assertStyleLayerOrder(raw);
    assertStyleLayerValidity(raw);
    // `dynamic` justifié : frontière vers l'API tierce `ThemeReader.read(Map<String, dynamic>)` ;
    // `jsonDecode` retourne `dynamic` et le renderer exige ce type exact (Pitfall 11).
    final Map<String, dynamic> json = jsonDecode(raw) as Map<String, dynamic>;
    return vtr.ThemeReader().read(json);
  }
}
